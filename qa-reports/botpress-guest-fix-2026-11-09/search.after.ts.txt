  '   Chất liệu: ' + (product.material || 'Chưa cập nhật'),
  '   URL chính thức: ' + (product.url || 'Không có URL xác thực'),
  '   Tình trạng các biến thể:',
  variantText
  ].join('\n')
  }).join('\n\n')
  workflow.productsText = /(?:bao nhieu|dem so|so luong)/.test(query) ? 'Có ' + workflow.productCount + ' sản phẩm phù hợp.\n\n' + productText : productText
  workflow.productsJson = JSON.stringify(processed.map((product: any) => ({
  id: product.id,
  name: product.name,
  price: product.minPrice === null ? '' : product.minPrice.toLocaleString('vi-VN'),
  minPrice: product.minPrice,
  description: product.description ?? '',
  material: product.material ?? '',
  slug: product.slug ?? '',
  url: product.url,
  gender_id: product.gender_id ?? 0,
  sizes: product.sizes,
  colors: product.colors,
  inStock: product.inStock,
  variants: product.variants,
  matchedKeyword: true,
  image: Array.isArray(product.image_url) ? (product.image_url[0] ?? '')
  : typeof product.image_url === 'string' ? product.image_url : ''
  })))
  }
  workflow.debugConsult = 'OK | products=' + processed.length +
  ' | linkRequest=' + wantsLink + ' | msg=' + userMessage
  } catch (err: any) {
  workflow.productCount = 0
  workflow.productsText = 'Hiện chưa thể truy vấn sản phẩm và tồn kho. Không tự suy đoán dữ liệu.'
  workflow.productsJson = '[]'
  workflow.debugConsult = 'ERR: ' + err.message
  }
